% Extract titles from a webpage
param(timeout, 'Timeout in ms', 5000).
param(max_results, 'Max results', 10).

% Fetch and extract titles
extract_titles(URL, Results) :-
  fetch_page(URL, Html),
  find_all(Html, 'h1', Titles),
  map_titles(Titles, Results).

site(homepage('https://example.com')).
